<style>.page__meta{display:none}</style>

{%- assign lang = page.lang
  | default: site.active_lang
  | default: site.default_lang
  | default: site.locale
  | default: 'en'
  | downcase | replace:'_','-' | split:'-' | first -%}


{%- for l in site.languages -%}
  {%- assign t = site.data['ui-text'][l].page_not_found_title | default: "404 - Page not found" -%}
  {%- assign i = site.data['ui-text'][l].page_not_found_info  | default: "The page you requested does not exist or has moved." -%}
  <div class="nf nf-{{ l }}" lang="{{ l }}">
    <h1>{{ t }}</h1>
    <p>{{ i }}</p>
  </div>
{%- endfor -%}


<script>
(function () {
  var langs = {{ site.languages | jsonify }};
  var defaultLang = "{{ site.default_lang | escape }}";
  var firstSeg = (location.pathname.split('/')[1] || "").toLowerCase();
  var active = (langs.indexOf(firstSeg) >= 0) ? firstSeg : defaultLang;

  var blocks = document.querySelectorAll('.nf');
  blocks.forEach(function(el){ el.hidden = true; });
  var activeEl = document.querySelector('.nf-' + active);
  if (activeEl) activeEl.hidden = false;
})();
</script>
